cmake_minimum_required(VERSION 3.22)
project(LindellStreamsLive VERSION 0.3.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(LSL_CORE_ONLY "Build dependency-free tests" OFF)
find_package(Threads REQUIRED)
enable_testing()
add_executable(lsl_core_tests Tests/CoreTests.cpp)
target_link_libraries(lsl_core_tests PRIVATE Threads::Threads)
add_test(NAME core COMMAND lsl_core_tests)
add_executable(lsl_link_tests Tests/PlaylistLinkTests.cpp)
add_test(NAME playlist_links COMMAND lsl_link_tests)
if(LSL_CORE_ONLY)
 return()
endif()
set(JUCE_PATH "" CACHE PATH "JUCE 8.0.15 source checkout")
set(AAX_SDK_PATH "" CACHE PATH "Locally licensed Avid AAX SDK")
option(LSL_BUILD_AAX "Build AAX Native for Emmanuel" OFF)
option(LSL_BUILD_TEST_HOST "Build plugin and media integration tests" ON)
add_subdirectory("${JUCE_PATH}" "${CMAKE_BINARY_DIR}/JUCE")
include(FetchContent)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(NO_EXAMPLES ON CACHE BOOL "" FORCE)
set(NO_TESTS ON CACHE BOOL "" FORCE)
set(NO_WEBSOCKET ON CACHE BOOL "" FORCE)
set(USE_MBEDTLS ON CACHE BOOL "" FORCE)
set(OPUS_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(OPUS_DISABLE_INTRINSICS ON CACHE BOOL "" FORCE)
set(LIBSAMPLERATE_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LIBSAMPLERATE_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(rtc GIT_REPOSITORY https://github.com/paullouisageneau/libdatachannel.git GIT_TAG v0.23.2 GIT_SHALLOW TRUE)
FetchContent_Declare(opus GIT_REPOSITORY https://github.com/xiph/opus.git GIT_TAG v1.5.2 GIT_SHALLOW TRUE)
FetchContent_Declare(samplerate GIT_REPOSITORY https://github.com/libsndfile/libsamplerate.git GIT_TAG 0.2.2 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(rtc opus samplerate)
if(APPLE)
 target_compile_options(srtp2 PRIVATE -Wno-error=shorten-64-to-32)
endif()
add_library(lsl_media INTERFACE)
target_link_libraries(lsl_media INTERFACE LibDataChannel::LibDataChannel Opus::opus samplerate Threads::Threads)
set(LSL_FORMATS VST3)
if(LSL_BUILD_AAX)
 juce_set_aax_sdk_path("${AAX_SDK_PATH}")
 list(APPEND LSL_FORMATS AAX)
endif()
juce_add_plugin(LindellStreamsLive
 COMPANY_NAME "Lindell Streams" BUNDLE_ID "com.lindellstreams.live.dev"
 PLUGIN_MANUFACTURER_CODE Lnds PLUGIN_CODE Lslv PRODUCT_NAME "Lindell Streams Live"
 FORMATS ${LSL_FORMATS} IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE
 COPY_PLUGIN_AFTER_BUILD FALSE)
set(LSL_SOURCES Source/PluginProcessor.cpp Source/PluginEditor.cpp Source/LiveSender.cpp)
target_sources(LindellStreamsLive PRIVATE ${LSL_SOURCES})
target_compile_definitions(LindellStreamsLive PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(LindellStreamsLive PRIVATE juce::juce_audio_utils lsl_media PUBLIC juce::juce_recommended_config_flags)
if(LSL_BUILD_TEST_HOST)
 juce_add_console_app(lsl_plugin_test)
 target_sources(lsl_plugin_test PRIVATE Tests/PluginSmoke.cpp ${LSL_SOURCES})
 target_compile_definitions(lsl_plugin_test PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
 target_link_libraries(lsl_plugin_test PRIVATE juce::juce_audio_utils lsl_media)
 add_executable(lsl_media_test Tests/MediaTest.cpp)
 target_link_libraries(lsl_media_test PRIVATE lsl_media)
 add_test(NAME codec_stereo COMMAND lsl_media_test)
endif()
